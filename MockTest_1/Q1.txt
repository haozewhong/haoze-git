class MovieTicket{
    private:
        string movieName;
        int quantity;
        double price;
        double total;

    public:
        MovieTicket(string name, int q, double p){
            movieName = name;
            quantity = q;
            price = p;
            total = 0.0;
        }
        void calculateTotal(){
            total = quantity * price;
        }
        void print(){
            cout << "Movie: " << movieName <<endl;
            cout << "Quantity: " << quantity <<endl;
            cout << "Total: RM" << total <<endl;
        }
};
