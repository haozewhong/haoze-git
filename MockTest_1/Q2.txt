class Product{
    private:
        string productName;
        int quantity;
        double price;
        double total;

    public:
        Product(string name, int q, double p){
            productName = name;
            quantity = q;
            price = p;
            total = 0.0;
        }
        void applyDiscount(double discount){
            total = (quantity * price) - discount;
            
        }
        void print(){
            cout << "Name: " << productName <<endl;
            cout << "Quantity: " << quantity <<endl;
            cout << "Final Total: RM" << total <<endl;
        }
};
